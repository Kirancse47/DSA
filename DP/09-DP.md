## Cheat sheet

### The five-step DP recipe
1. Express the problem as "state -> answer for that state".
2. Write the recurrence (transition) from a state to smaller states.
3. Identify the base cases.
4. Memoize the recursion (top-down), or fill a table bottom-up (tabulation).
5. Drop table dimensions you no longer read (space optimization).

Converting memoization -> tabulation: reverse the recursion's direction, turn each
recursive parameter into a loop, and make the loop order respect the dependency.
Converting tabulation -> 1-D: if `dp[i][*]` only ever reads `dp[i-1][*]`, keep two
rows (`prev`, `cur`); if the reads are all at lower indices, a single array iterated
in the right direction is enough.

### Patterns in this directory

| Pattern | Signature recurrence | Problems here |

| Unbounded knapsack | `dp[i][w] = max(val[i]+dp[i][w-wt[i]], dp[i-1][w])` | Rod Cutting, 322 Coin Change, 518 Coin Change 2, 377 Combination Sum IV |
| LCS family (2 strings) | `dp[i][j] = s1[i]==s2[j] ? 1+dp[i-1][j-1] : max(dp[i-1][j], dp[i][j-1])` | 1143 LCS, 72 Edit Distance, 1092 Shortest Common Supersequence, Longest Common Substring, 115 Distinct Subsequences |

### Complexity rule of thumb
Time = (number of distinct states) x (work per transition).
Space = table size, plus `O(recursion depth)` stack for the memoized version.

## 1. 0 - 1 Knapsack Problem

**Q:** Find the maximum total value of items that can be put in a knapsack of capacity `W`, given the weights `wt` and values `val` of `n` items, where each item can be chosen at most once.

**A:** 2-D bottom-up DP over (remaining items, remaining capacity).

**Explanation:**
- `dp[i][j]` represents the maximum value achievable using the first `i` items with a remaining capacity `j`.
- We transition by either taking the `i-1`-th item if it fits (`val[i-1] + dp[i-1][j-wt[i-1]]`) or skipping it (`dp[i-1][j]`), taking the maximum of both.
- The base case is `dp[0][j] = 0` and `dp[i][0] = 0` (no items or no capacity yields zero value).
- The final answer is located at `dp[n][W]`.

**Complexity:**
Memoization: Time `O(n*W)`, Space `O(n*W)` + `O(n)` stack
Tabulation: Time `O(n*W)`, Space `O(n*W)`
Space optimized: Time `O(n*W)`, Space `O(W)`
More Space optimized: Time `O(n*W)`, Space `O(W)`

```cpp
// Memorization.....
    vector<vector<int>> dp;
    int solve(int W, int wt[], int val[], int n){
      if(n==0 || W==0) return 0;
      if(dp[n][W]!=-1) return dp[n][W];
      if(W<wt[n-1]) return dp[n][W]=solve(W,wt,val,n-1);
      return dp[n][W]=max(val[n-1]+solve(W-wt[n-1],wt,val,n-1),solve(W,wt,val,n-1));    
    }
    int knapSack(int W, int wt[], int val[], int n){ 
        dp=vector<vector<int>>(n+1,vector<int>(W+1,-1));
        return solve(W,wt,val,n);
    }
    
// Tabulation....
    int knapSack(int W, int wt[], int val[], int n){
        vector<vector<int>> dp(n+1,vector<int>(W+1,0));
        for(int i=0;i<n+1;i++){
            for(int j=0;j<W+1;j++){
                if(i==0 || j==0) dp[i][j]=0;
                else if(j>=wt[i-1]) dp[i][j]=max(val[i-1]+dp[i-1][j-wt[i-1]],dp[i-1][j]);
                else dp[i][j]=dp[i-1][j];
            }
        }
        return dp[n][W];
    }
    
// Space optimized....
    int knapSack(int W, int wt[], int val[], int n){
        vector<int> prev(W+1,0), cur(W+1,0);
        for(int i=0;i<n+1;i++){
            for(int j=0;j<W+1;j++){
                if(i==0 || j==0) cur[j]=0;
                else if(j>=wt[i-1]) cur[j]=max(val[i-1]+prev[j-wt[i-1]],prev[j]);
                else cur[j]=prev[j];
            }
            prev=cur;
        }
        return prev[W];
    }
    
// More Space optimized....
    int knapSack(int W, int wt[], int val[], int n){
        vector<int> dp(W+1,0);
        for(int i=1;i<n+1;i++){
            for(int j=W;j>=0;j--){
                if(j>=wt[i-1]) dp[j]=max(val[i-1]+dp[j-wt[i-1]],dp[j]);
            }
        }
        return dp[W];
    }
```

## 1043. Partition Array for Maximum Sum

**Q:** Partition an integer array `arr` of size `n` into contiguous subarrays of maximum length `k` such that each element in a subarray is changed to the maximum value of that subarray, maximizing the total sum.

**A:** 1-D DP over the starting index of the partition.

**Explanation:**
- `dp[i]` represents the maximum sum we can get by partitioning the suffix of `arr` starting from index `i`.
- We transition by iterating the end of the current partition `j` from `i` up to `min(n-1, i+k-1)`, keeping track of the max element `mxe`, and updating the sum with `(j-i+1)*mxe + dp[j+1]`.
- The base case is `dp[n] = 0` because an empty suffix has a sum of 0.
- The final answer is read from `dp[0]`.

**Complexity:**
Memoization: Time `O(n*k)`, Space `O(n)` + `O(n)` stack
Tabulation: Time `O(n*k)`, Space `O(n)`

```cpp
// Memoization...
    int solve(int i,int n,vector<int> &a,int k,vector<int> &dp){
        if(i==n) return 0;
        if(dp[i]!=-1) return dp[i];
        int ans=INT_MIN;
        int l=0,mxe=INT_MIN;
        for(int j=i;j<min(n,i+k);j++){
            l++;
            mxe=max(mxe,a[j]);
            ans=max(ans,l*mxe+solve(j+1,n,a,k,dp));
        }
        return dp[i]=ans;
    }
    
    int maxSumAfterPartitioning(vector<int>& arr, int k) {
        int n=arr.size();
        vector<int> dp(n,-1);
        return solve(0,n,arr,k,dp);
    }
    
// Tabulation...
     int maxSumAfterPartitioning(vector<int>& arr, int k) {
        int n=arr.size();
        vector<int> dp(n+1,0);
        for(int i=n-1;i>=0;i--){
            int ans=INT_MIN;
            int l=0,mxe=INT_MIN;
            for(int j=i;j<min(n,i+k);j++){
                l++;
                mxe=max(mxe,arr[j]);
                ans=max(ans,l*mxe+dp[j+1]);
            }
            dp[i]=ans;
        }
         return dp[0];
    }
```

## 1048. Longest String Chain

**Q:** Find the length of the longest possible word chain among a given list of `words`, where a word chain is formed by inserting exactly one letter into a word to get the next word.

**A:** 1-D DP over the length-sorted words list.

**Explanation:**
- `dp[i]` represents the length of the longest word chain ending at `words[i]`.
- We transition by comparing `words[i]` against all previous words `words[j]` (where `j < i`). If `words[j]` is a predecessor of `words[i]`, we update `dp[i] = max(dp[i], 1 + dp[j])`.
- The base case is `dp[i] = 1` for all `i`, representing a chain of length 1 for the word itself.
- The final answer is the maximum value across the entire `dp` array.

**Complexity:**
Time `O(n^2 * L + n log n)`, where `n` is the number of words and `L` is the maximum length of a word.
Space `O(n)`

```cpp
    bool check(string &si,string &sj){
        int li=si.size(),lj=sj.size();
        if(li-lj!=1) return false;
        int i=0,j=0;
        while(i<li){
            if(si[i]==sj[j]){
                i++;j++;
            }else{
                i++;
            }
        }
        if(i==li && j==lj) return true; 
        else return false;
    }
    
    static bool cmp(string &s1,string &s2){
        return s1.size()<s2.size();
    }
    
    int longestStrChain(vector<string>& words) {
        int n=words.size();
        sort(words.begin(),words.end(),cmp);
        vector<int> dp(n,1);
        int ans=1;
        for(int i=1;i<n;i++){
            for(int j=i-1;j>=0;j--){
                if(check(words[i],words[j]) && 1+dp[j]>dp[i]){
                    dp[i]=1+dp[j];
                }
            }
            ans=max(ans,dp[i]);
        }
        return ans;
    }
```

## 1092. Shortest Common Supersequence

**Q:** Find the shortest string that has both `str1` and `str2` as subsequences.

**A:** DP to find the Longest Common Subsequence (LCS) matrix, followed by string building using the DP grid.

**Explanation:**
- `dp[i][j]` stores the length of the LCS of `str1[0..i-1]` and `str2[0..j-1]`.
- We transition by extending the LCS if `str1[i-1] == str2[j-1]` or inheriting the maximum length from `dp[i-1][j]` or `dp[i][j-1]`.
- Using the `dp` table, we trace backwards: if characters match, we include them once; otherwise, we include the character from the string corresponding to the larger `dp` value.
- The answer is the reverse of the constructed string.

**Complexity:**
Time `O(n1*n2)`, Space `O(n1*n2)`

```cpp
string shortestCommonSupersequence(string str1, string str2) {
        int n1=str1.size(),n2=str2.size();
        vector<vector<int>> dp(n1+1,vector<int>(n2+1,0));
        for(int i=1;i<n1+1;i++){
            for(int j=1;j<n2+1;j++){
                if(str1[i-1]==str2[j-1]) dp[i][j]=1+dp[i-1][j-1];
                else dp[i][j]=max(dp[i-1][j],dp[i][j-1]);
            }
        }
        // len=n1+n2-lcs(str1,str2);
        
        string ans="";
        int i=n1,j=n2;
        while(i>0 && j>0){
            if(str1[i-1]==str2[j-1]){
                ans.push_back(str1[i-1]);
                i--;j--;
            }else{
                if(dp[i][j-1]>dp[i-1][j]){
                    ans.push_back(str2[j-1]);
                    j--;
                }else{
                    ans.push_back(str1[i-1]);
                    i--;
                }
            }
        }
        while(i>0){
            ans.push_back(str1[i-1]);
            i--;
        }
        while(j>0){
            ans.push_back(str2[j-1]);
            j--;
        }
        reverse(ans.begin(),ans.end());
        return ans;
    }
```

## 10. Regular Expression Matching

**Q:** Determine if a string `s` matches a regular expression pattern `p` containing '.' (any single character) and '*' (zero or more of the preceding element).

**A:** 2-D DP matching prefixes of `s` and `p`.

**Explanation:**
- `dp[i][j]` is a boolean indicating if `s[0..i-1]` matches `p[0..j-1]`.
- We transition primarily based on whether `p[j-1]` is a '*'. If it is, we can either ignore the '*' and preceding character (`dp[i][j-2]`), or use the '*' if the preceding characters match (`dp[i-1][j]`).
- The base case is `dp[0][0] = true` and `dp[i][0] = false`. `dp[0][j]` depends on whether `p[j-1]` is '*' and `dp[0][j-2]` is true.
- The final answer is found at `dp[n][m]`.

**Complexity:**
Recursive: Time `O(2^(n+m))` (worst case), Space `O(n+m)` stack
Memoization: Time `O(n*m)`, Space `O(n*m)` + `O(n+m)` stack

```cpp
// Recursive..
    bool solve(int i,int j,string &s,string &p){
        if(i==0 && j==0) return true;
        if(j==0) return false;
        if(i==0){
            if(p[j-1]=='*') return solve(i,j-2,s,p);
            else return false;
        }
        if(s[i-1]==p[j-1] || p[j-1]=='.') return solve(i-1,j-1,s,p);
        if(p[j-1]=='*'){
            if(s[i-1]==p[j-2] || p[j-2]=='.') return solve(i-1,j,s,p) || solve(i,j-2,s,p);
            else return solve(i,j-2,s,p);
        }
        return false;
    }
    
    bool isMatch(string s, string p) {
        int n=s.size(),m=p.size();
        return solve(n,m,s,p);
    }
   
    
// Memoization...
    bool solve(int i,int j,string &s,string &p,vector<vector<int>> &dp){
        if(i==0 && j==0) return true;
        if(j==0) return false;
        if(i==0){
            if(p[j-1]=='*') return solve(i,j-2,s,p,dp);
            else return false;
        }
        if(dp[i][j]!=-1) return dp[i][j];
        if(s[i-1]==p[j-1] || p[j-1]=='.') return dp[i][j]=solve(i-1,j-1,s,p,dp);
        if(p[j-1]=='*'){
            if(s[i-1]==p[j-2] || p[j-2]=='.') return dp[i][j]=solve(i-1,j,s,p,dp) || solve(i,j-2,s,p,dp);
            else return dp[i][j]=solve(i,j-2,s,p,dp);
        }
        return dp[i][j]=false;
    }
    
    bool isMatch(string s, string p) {
        int n=s.size(),m=p.size();
        vector<vector<int>> dp(n+1,vector<int>(m+1,-1));
        return solve(n,m,s,p,dp);
    }
```

## 1143. Longest Common Subsequence

**Q:** Find the length of the longest common subsequence between two strings `text1` and `text2`.

**A:** 2-D DP over the prefixes of both strings.

**Explanation:**
- `dp[i][j]` represents the length of the LCS for the prefixes `text1[0..i-1]` and `text2[0..j-1]`.
- We transition by adding 1 to `dp[i-1][j-1]` if the current characters match (`text1[i-1] == text2[j-1]`). Otherwise, we take the max of excluding the current character from either string: `max(dp[i-1][j], dp[i][j-1])`.
- The base case is `dp[0][j] = 0` and `dp[i][0] = 0`, as the LCS with an empty string is always 0.
- The final answer is read from `dp[n1][n2]`.

**Complexity:**
Memoization: Time `O(x*y)`, Space `O(x*y)` + `O(x+y)` stack
Tabulation: Time `O(n1*n2)`, Space `O(n1*n2)`
Space optimize: Time `O(n1*n2)`, Space `O(n2)`

```cpp
// Memorization.....
    int solve(int x, int y, string &s1, string &s2,vector<vector<int>> &dp){
        if(x==0 || y==0) return 0;
        if(dp[x][y]!=-1) return dp[x][y];
        if(s1[x-1]==s2[y-1]) return dp[x][y]=1+solve(x-1,y-1,s1,s2,dp);
        else return dp[x][y]=max(solve(x-1,y,s1,s2,dp),solve(x,y-1,s1,s2,dp));
    }
    
    int longestCommonSubsequence(string text1, string text2) {
        int x=text1.size(),y=text2.size();
        vector<vector<int>> dp(x+1,vector<int>(y+1,-1));
        return solve(x,y,text1,text2,dp);
    }

//Tabulation......
    int longestCommonSubsequence(string &s1, string &s2) {
        int n1=s1.size(),n2=s2.size();
        vector<vector<int>> dp(n1+1,vector<int>(n2+1,0));
        for(int i=1;i<n1+1;i++){
            for(int j=1;j<n2+1;j++){
                if(s1[i-1]==s2[j-1]) dp[i][j]=1+dp[i-1][j-1];
                else dp[i][j]=max(dp[i-1][j],dp[i][j-1]);
            }
        }
        return dp[n1][n2];
    }
    
// Space optimize....
    int longestCommonSubsequence(string &s1, string &s2) {
        int n1=s1.size(),n2=s2.size();
        vector<int> prev(n2+1,0),cur(n2+1,0);
        for(int i=1;i<n1+1;i++){
            for(int j=1;j<n2+1;j++){
                if(s1[i-1]==s2[j-1]) cur[j]=1+prev[j-1];
                else cur[j]=max(prev[j],cur[j-1]);
            }
            prev=cur;
        }
        return prev[n2];
    }
```

## 115. Distinct Subsequences

**Q:** Count the number of distinct subsequences of string `s` that equal string `t`.

**A:** 2-D DP over the prefixes of `s` and `t`.

**Explanation:**
- `dp[i][j]` represents the number of distinct subsequences of `s[0..i-1]` that match `t[0..j-1]`.
- We transition by carrying over the matches ignoring the current character of `s` (`dp[i-1][j]`). If the current characters match (`s[i-1] == t[j-1]`), we also add the matches formed by including this character (`dp[i-1][j-1]`).
- The base case is `dp[i][0] = 1` (an empty `t` can be formed by 1 subsequence: the empty one) and `dp[0][j] = 0` for `j > 0`.
- The final answer is read from `dp[n][m]`.

**Complexity:**
Memoization: Time `O(n*m)`, Space `O(n*m)` + `O(n+m)` stack
Tabulation: Time `O(n*m)`, Space `O(n*m)`
Space Optimize: Time `O(n*m)`, Space `O(m)`
More Space Optimize: Time `O(n*m)`, Space `O(m)`

```cpp
 int mod=1e9+7;
    
// Memorization...
    int solve(int n,int m,string s,string t,vector<vector<int>> &dp){
        if(m==0) return 1;
        if(n==0) return 0;
        if(dp[n][m]!=-1) return dp[n][m];
        if(s[n-1]==t[m-1]) return dp[n][m]=( solve(n-1,m-1,s,t,dp) + solve(n-1,m,s,t,dp) )%mod;
        else return dp[n][m]=(solve(n-1,m,s,t,dp))%mod;
    }
     
    int numDistinct(string s, string t) {
        int n=s.size(),m=t.size();
        vector<vector<int>> dp(n+1,vector<int>(m+1,-1));
        return solve(n,m,s,t,dp);
    }
    
// Tabulation...
    int numDistinct(string s, string t) {
      int n=s.size(),m=t.size();
      vector<vector<int>> dp(n+1,vector<int>(m+1,0));
      for(int i=0;i<n+1;i++) dp[i][0]=1;
      
      for(int i=1;i<n+1;i++){
          for(int j=1;j<m+1;j++){
              if(s[i-1]==t[j-1]) dp[i][j]=(dp[i-1][j-1]+dp[i-1][j])%mod;
              else dp[i][j]=dp[i-1][j]%mod;
          }
      }
      return dp[n][m];
    }
    
// Space Optimize...
    int numDistinct(string s, string t) {
      int n=s.size(),m=t.size();
      vector<int> prev(m+1,0),cur(m+1,0);
      prev[0]=1; cur[0]=1;
      
      for(int i=1;i<n+1;i++){
          for(int j=1;j<m+1;j++){
              if(s[i-1]==t[j-1]) cur[j]=(prev[j-1]+prev[j])%mod;
              else cur[j]=prev[j] % mod;
          }
          prev=cur;
      }
      return prev[m];
    }
    
// More Space Optimize...
    int numDistinct(string s, string t) {
      int n=s.size(),m=t.size();
      vector<int> dp(m+1,0);
      dp[0]=1;
      
      for(int i=1;i<n+1;i++){
          for(int j=m;j>0;j--){
              if(s[i-1]==t[j-1]) dp[j]=(dp[j-1]+dp[j])%mod;
          }
      }
      return dp[m];
    }
```

## 120. Triangle

**Q:** Find the minimum path sum from the top to the bottom of a `triangle` array, moving only to adjacent numbers on the row below.

**A:** 2-D bottom-up DP starting from the base of the triangle.

**Explanation:**
- `dp[i][j]` represents the minimum path sum starting from `triangle[i][j]` to the bottom row.
- We transition by adding `triangle[i][j]` to the minimum path sum of the two adjacent elements in the row below: `min(dp[i+1][j], dp[i+1][j+1])`.
- The base case initializes the bottom row `dp[n-1][j] = triangle[n-1][j]`.
- The final answer is read from `dp[0][0]`.

**Complexity:**
Recursive: Time `O(2^n)`, Space `O(n)` stack
Memoization: Time `O(n^2)`, Space `O(n^2)` + `O(n)` stack
Tabulation: Time `O(n^2)`, Space `O(n^2)`
Space Optimaize: Time `O(n^2)`, Space `O(n)`

```cpp
 // recursive t-O(2^n) s-O(n)...
    int solve(int i,int j,int n,vector<vector<int>>& t){
        if(i==n-1) return t[i][j];
        int down=t[i][j]+solve(i+1,j,n,t);
        int dia=t[i][j]+solve(i+1,j+1,n,t);
        return min(down,dia);
    }
    int minimumTotal(vector<vector<int>>& triangle) {
          int n=triangle.size();
        return solve(0,0,n,triangle);
    }

        if(dp[i][j]!=-1) return dp[i][j];
        int down=t[i][j]+solve(i+1,j,n,t,dp);
        int dia=t[i][j]+solve(i+1,j+1,n,t,dp);
        return dp[i][j]=min(down,dia);
```

## 122. Best Time to Buy and Sell Stock II

**Q:** Find the maximum profit achievable by completing as many transactions as you like (buy one and sell one share of the stock multiple times).

- The final answer is read from `dp[0][1]` (starting on day 0 with the ability to buy).

## 1277. Count Square Submatrices with All Ones

**Q:** Count how many square submatrices have all ones in a given `m x n` binary `matrix`.

**A:** 2-D DP where the state represents the side length of the maximal square whose bottom-right corner is at a given cell.

**Explanation:**
- `dp[i][j]` represents the side length of the largest square submatrix ending at cell `(i, j)`. It also exactly equals the number of square submatrices ending at `(i, j)`.
- We transition by checking if `matrix[i][j] == 1`. If so, we take the minimum of the three adjacent top/left DP values and add 1: `dp[i][j] = 1 + min(dp[i-1][j-1], min(dp[i-1][j], dp[i][j-1]))`.
- The base case is the first row and first column, where `dp[i][j]` is just the matrix value.
- The final answer is the sum of all elements in the `dp` grid.

**Complexity:**
Time `O(n*m)`, Space `O(n*m)`

```cpp
    int countSquares(vector<vector<int>>& matrix) {
        int n=matrix.size(), m=matrix[0].size();
        vector<vector<int>> dp(n,vector<int>(m,0));
        int ans=0;
        for(int i=0;i<n;i++){
            dp[i][0]=matrix[i][0];
            ans+=dp[i][0];
        }
        for(int i=1;i<m;i++){
            dp[0][i]=matrix[0][i];
            ans+=dp[0][i];
        }
        for(int i=1;i<n;i++){
            for(int j=1;j<m;j++){
                if(matrix[i][j]){
                    dp[i][j]=1+min(dp[i-1][j-1],min(dp[i-1][j],dp[i][j-1]));
                    ans+=dp[i][j];
                }
            }
        }
        return ans;
    }
```

## 1312. Minimum Insertion Steps to Make a String Palindrome

**Q:** Find the minimum number of character insertions required to make a string `s` a palindrome.

**A:** 2-D DP to find the Longest Palindromic Subsequence (LPS), derived via LCS with the reversed string.

**Explanation:**

- The transition for LCS uses `cur[j] = 1 + prev[j-1]` on match and `cur[j] = max(cur[j-1], prev[j])` otherwise.
- The base case for LCS is initialized to 0 for length 0 strings.
- The final answer is the length of the string `s` minus the length of its LPS, representing the characters lacking a matching pair.

**Complexity:**

```cpp
   int lps(string s){

            for(int j=1;j<n+1;j++){
                if(s[i-1]==s[n-j]) cur[j]=1+prev[j-1];
                else cur[j]=max(cur[j-1],prev[j]);
            }
            prev=cur;
        }
        return prev[n];
    }
```

## 198. House Robber

**Q:** Find the maximum amount of money you can rob from an array of houses `nums` without robbing two adjacent houses.

**A:** 1-D DP comparing the choice of robbing or skipping the current house.

**Explanation:**
- `dp[i]` represents the maximum money robbed from the prefix of houses up to index `i`.
- We transition by deciding to either rob house `i` (adding its value to `dp[i-2]`) or skip it (keeping `dp[i-1]`), choosing the max: `dp[i] = max(nums[i] + dp[i-2], dp[i-1])`.
- The base case is `dp[0] = nums[0]`.
- The final answer is read from `dp[n-1]`.

## 322. Coin Change (Min no of coins))

**Q:** Given an array of coin denominations and a target amount, return the minimum number of coins needed to make up that amount (or -1 if impossible).

**A:** Unbounded knapsack DP for minimum elements.

**Explanation:**
- The DP state `dp[i][j]` tracks the minimum coins to form amount `j` using the first `i` coin types.
- For each coin, we can either exclude it (`dp[i-1][j]`) or include it (`1 + dp[i][j-coins[i-1]]`).

- Space optimizations reduce the table to just the current and previous rows, or even a single 1D array by processing left-to-right.

Memorization: Time `O(n * amount)`, Space `O(n * amount)` + `O(amount)` stack
Tabulation: Time `O(n * amount)`, Space `O(n * amount)`

## 368. Largest Divisible Subset

**Q:** Given a set of distinct positive integers, find the largest subset such that every pair `(nums[i], nums[j])` satisfies `nums[i] % nums[j] == 0` or `nums[j] % nums[i] == 0`.

**A:** Longest Increasing Subsequence (LIS) style DP with parent pointers to reconstruct the subset.

**Explanation:**
- The array is sorted first, so we only need to check if `nums[i] % nums[j] == 0` for `j < i`.
- The DP state `dp[i]` represents the length of the largest divisible subset ending at index `i`.
- `idx[i]` acts as a parent pointer array to track the previous element in the subset for reconstruction.
- After filling the DP table and tracking the maximum length's end index `lastIdx`, we backtrack through `idx` to build the actual subset.

**Complexity:**
Tabulation: Time `O(n^2)`, Space `O(n)`

```cpp
    vector<int> largestDivisibleSubset(vector<int>& nums) {
        int n=nums.size();
        sort(nums.begin(),nums.end());
        vector<int> dp(n,1),idx(n);
        int maxl=1,lastIdx=0;
        for(int i=1;i<n;i++){
            idx[i]=i;
            for(int j=i-1;j>=0;j--){
                if(nums[i]%nums[j]==0 && 1+dp[j]>dp[i]){
                    dp[i]=1+dp[j];
                    idx[i]=j;
                }
            }
            if(dp[i]>maxl){
                maxl=dp[i];
                lastIdx=i;
            }
        }
        vector<int>ans;
        ans.push_back(nums[lastIdx]);
        while(idx[lastIdx]!=lastIdx){
            lastIdx=idx[lastIdx];
            ans.push_back(nums[lastIdx]);
        }
        reverse(ans.begin(),ans.end());
        return ans;
    }
```

## 376. Wiggle Subsequence

**Q:** Given an integer array, return the length of the longest wiggle subsequence (where differences between consecutive numbers alternate between positive and negative).

**A:** 1D DP tracking the length of the longest wiggle sequence ending with an 'up' or 'down' move.

**Explanation:**
- We maintain two state variables: `up` (length of the longest wiggle ending with a positive difference) and `down` (length of the longest wiggle ending with a negative difference).
- If `nums[i] < nums[i-1]`, it's a down-move, so `down = up + 1`.
- If `nums[i] > nums[i-1]`, it's an up-move, so `up = down + 1`.
- We return the maximum of `up` and `down` at the end.

**Complexity:**
Tabulation: Time `O(n)`, Space `O(1)`

```cpp
    int wiggleMaxLength(vector<int>& nums) {
        int n=nums.size();
        int up=1,down=1;
        for(int i=1;i<n;i++){
            if(nums[i]<nums[i-1]) down=up+1;
            if(nums[i]>nums[i-1]) up=down+1;
        }
        return max(up,down);
    }
```

## 377. Combination Sum IV

**Q:** Given an array of distinct integers `nums` and a target integer `target`, return the number of possible combinations that add up to `target`.

**A:** Unbounded knapsack counting paths (permutations allowed).

**Explanation:**
- The DP state `dp[tar]` represents the number of ways to form the target sum `tar`.
- We iterate through every number in `nums` and if it is `<= tar`, we add `dp[tar - num]` to `dp[tar]`.
- The base case is `dp[0] = 1` because there is exactly 1 way to form the sum 0 (by choosing nothing).
- The final answer is found at `dp[target]`.

**Complexity:**
Recursive: Time `O(target^target)`, Space `O(target)` stack

```cpp
        return ans;
```

## 486. Predict the Winner

**Q:** Given an integer array `nums`, return true if player 1 can win a game where players alternately pick numbers from either end of the array to maximize their score.

**A:** Minimax DP over intervals `[i, j]`.

**Explanation:**
- The DP state `dp[i][j]` computes the maximum possible score Player 1 can guarantee from the subarray `[i, j]`.
- Transition simulates the opponent playing optimally to minimize Player 1's gain: `max(nums[i] + min(dp[i+2][j], dp[i+1][j-1]), nums[j] + min(dp[i+1][j-1], dp[i][j-2]))`.
- The base case for length 1 (`i == j`) is just `nums[i]`.
- We return true if the maximum score guaranteed for Player 1 is at least half the total sum of the array.

**Complexity:**
Recursion: Time `O(2^n)`, Space `O(n)` stack
Memoization: Time `O(n^2)`, Space `O(n^2)` + `O(n)` stack

```cpp
// Recursion...
    int help(vector<int>& nums,int i,int j){
        if(i>j) return 0;
        if(i==j) return nums[i];
        int ci=nums[i]+min(help(nums,i+2,j),help(nums,i+1,j-1));
        int cj=nums[j]+min(help(nums,i+1,j-1),help(nums,i,j-2));
        return max(ci,cj);
    }
    
    bool PredictTheWinner(vector<int>& nums) {
        int sum=accumulate(nums.begin(),nums.end(),0);
        int ans=help(nums,0,nums.size()-1);
        return ans>=(sum-ans);
    }

// Memoization...

        if(i==j) return nums[i];
        if(dp[i][j]!=-1) return dp[i][j];
        int ci=nums[i]+min(help(nums,dp,i+2,j),help(nums,dp,i+1,j-1));
        int cj=nums[j]+min(help(nums,dp,i+1,j-1),help(nums,dp,i,j-2));
        return dp[i][j]=max(ci,cj);
    }

        int sum=accumulate(nums.begin(),nums.end(),0);
        int n=nums.size();
        vector<vector<int>> dp(n+1,vector<int>(n+1,-1));
        int ans=help(nums,dp,0,n-1);

    }
```

**Explanation:**

- If characters match (`s[i-1] == rs[j-1]`), `dp[i][j] = 1 + dp[i-1][j-1]`.
- If they don't match, `dp[i][j] = max(dp[i-1][j], dp[i][j-1])`.
- The answer is found at `dp[n][n]`.

**Complexity:**
Tabulation: Time `O(n^2)`, Space `O(n^2)`
Space Optimize: Time `O(n^2)`, Space `O(n)`

```cpp
        int n=s.size();
        string rs=s;
        reverse(rs.begin(),rs.end());
        vector<vector<int>> dp(n+1,vector<int>(n+1,0));
        for(int i=1;i<n+1;i++){
            for(int j=1;j<n+1;j++){
                if(s[i-1]==rs[j-1]){
                    dp[i][j]=1+dp[i-1][j-1];
                }else{
                    dp[i][j]=max(dp[i-1][j],dp[i][j-1]);
                }
            }
        }
        return dp[n][n];
    }
```

## 518. Coin Change 2 (No of ways)

**Q:** Given an array of coin denominations and an integer amount, return the total number of combinations that make up that amount.

**A:** Unbounded knapsack counting combinations.

**Explanation:**
- The DP state `dp[n][amt]` stores the number of ways to make the amount `amt` using the first `n` coin types.
- For each coin, we sum the ways including it (`dp[n][amt-coins[n-1]]`) and the ways excluding it (`dp[n-1][amt]`).
- Base cases: 1 way to make amount 0 (`dp[n][0] = 1`) and 0 ways if no coins remain for `amt > 0`.
- The space is optimized down to a single row since the current row only depends on itself and the previous row.

**Complexity:**
Memoization: Time `O(n * amount)`, Space `O(n * amount)` + `O(amount)` stack
Tabulation: Time `O(n * amount)`, Space `O(n * amount)`
Space Optimized: Time `O(n * amount)`, Space `O(amount)`
More Space Optimized: Time `O(n * amount)`, Space `O(amount)`

```cpp

        if(amt>=coins[n-1]) return dp[n][amt]=solve(n,amt-coins[n-1],coins,dp)+solve(n-1,amt,coins,dp);
```

## 576. Out of Boundary Paths

**Q:** Given an `m x n` grid and a start coordinate, return the number of paths to move a ball out of the grid boundary with at most `maxMove` moves.

**A:** 3D DP tracking coordinates and remaining moves.

**Explanation:**
- The DP state `dp[i][j][mv]` represents the number of ways to reach out of bounds from `(i, j)` with exactly `mv` moves remaining.
- We transition by moving in all four cardinal directions, summing `solve(i+dx, j+dy, mv-1)`.
- Base cases: if out of bounds (`i<0` or `j<0` or `i==m` or `j==n`), return 1 path found. If `mv == 0` while still inside, return 0.
- Since answers can be large, results are taken modulo `1e9+7`.

**Complexity:**
Recursive: Time `O(4^N)`, Space `O(N)` stack
Memoization: Time `O(m * n * maxMove)`, Space `O(m * n * maxMove)` + `O(maxMove)` stack

```cpp
// Recurssive....
    int findPaths(int m, int n, int N, int i, int j) {
        if (i == m || j == n || i < 0 || j < 0) return 1;
        if (N == 0) return 0;
        return findPaths(m, n, N - 1, i - 1, j)
            + findPaths(m, n, N - 1, i + 1, j)
            + findPaths(m, n, N - 1, i, j - 1)
            + findPaths(m, n, N - 1, i, j + 1);
    }
    
// Memoization...
    int mod=1e9+7;
    int solve(int i,int j,int mv,int m,int n,vector<vector<vector<int>>> &dp){

        if(mv==0)  return 0;
        if(dp[i][j][mv]!=-1 ) return dp[i][j][mv];
        int dx[] = {-1, 0, 1, 0};
        int dy[] = {0, 1, 0, -1};
        int ans=0;
        for(int k=0;k<4;k++){
            ans=(ans+solve(i+dx[k],j+dy[k],mv-1,m,n,dp))%mod;
        }
        return dp[i][j][mv]=ans;
    }
    
    int findPaths(int m, int n, int maxMove, int startRow, int startColumn) {
        vector<vector<vector<int>>> dp(m,vector<vector<int>>(n,vector<int>(maxMove+1,-1)));
        return solve(startRow,startColumn,maxMove,m,n,dp);
    }
```

## 62. Unique Paths

**Q:** Given an `m x n` grid, return the number of unique paths from the top-left corner to the bottom-right corner, moving only down or right.

**A:** 2D DP counting paths by summing paths from the top and left neighbors.

**Explanation:**
- The DP state `dp[i][j]` tracks the number of unique paths to reach cell `(i, j)`.
- The transition is `dp[i][j] = dp[i-1][j] + dp[i][j-1]` because you can only arrive from above or from the left.
- Base case is `dp[0][0] = 1`, and out of bounds indices yield 0.
- Space optimization stores only the previous row (`prev`) and computes the current row (`cur`) progressively.

**Complexity:**
Recursive: Time `O(2^(m+n))`, Space `O(m+n)` stack
Memorization: Time `O(m * n)`, Space `O(m * n)` + `O(m + n)` stack
Tabulation: Time `O(m * n)`, Space `O(m * n)`
Space optimization: Time `O(m * n)`, Space `O(n)`

```cpp
//Recursive....O(2^mn) O(m+n)
    int solve(int i,int j){
        if(i==0 && j==0) return 1;
        if(i<0 || j<0) return 0;
        return solve(i-1,j)+solve(i,j-1);
    }
    int uniquePaths(int m, int n) {
        return solve(m-1,n-1);
    }
    
//Memorization....O(mn) O(mn)
    int solve(int i,int j,vector<vector<int>> &dp){
        if(i==0 && j==0) return 1;
        if(i<0 || j<0) return 0;
        if(dp[i][j]!=-1) return dp[i][j];
        return dp[i][j]=solve(i-1,j,dp)+solve(i,j-1,dp);
    }
    int uniquePaths(int m, int n) {
        vector<vector<int>> dp(m,vector<int>(n,-1));
        return solve(m-1,n-1,dp);
    }
    
//Tabulation...O(mn) O(mn)

        vector<vector<int>> dp(m,vector<int>(n,1));
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                dp[i][j]= (i>0 ? dp[i-1][j] : 0) + (j>0 ? dp[i][j-1] : 0);
            }
        }
        return dp[m-1][n-1];        
    }
    
//Space optimization...O(mn) O(n)

        vector<int> prev(n,1);
        for(int i=1;i<m;i++){
            vector<int> cur(n,1);
            for(int j=1;j<n;j++){
                cur[j]=cur[j-1]+prev[j];
            }
            prev=cur;
        }
        return prev[n-1];
```

## 64. Minimum Path Sum

**Q:** Given an `m x n` grid filled with non-negative numbers, return a path from top-left to bottom-right that minimizes the sum of all numbers along its path.

**A:** 2D DP minimizing accumulated sum from top and left neighbors.

**Explanation:**
- The DP state `dp[i][j]` holds the minimum path sum to reach cell `(i, j)`.
- The transition takes the current cell's value plus the minimum cost of reaching it from above or left: `g[i][j] + min(dp[i-1][j], dp[i][j-1])`.
- Base cases: `dp[0][0] = g[0][0]`, and out-of-bounds accesses return infinity (`1e9`).
- Space optimization retains only a 1D array representing the previous row.

**Complexity:**
Memoization: Time `O(m * n)`, Space `O(m * n)` + `O(m + n)` stack
Tabulation: Time `O(m * n)`, Space `O(m * n)`
Space Optimized: Time `O(m * n)`, Space `O(n)`

```cpp
// Memoization...
    int solve(int i,int j, vector<vector<int>>& g,vector<vector<int>>& dp){
        if(i==0 && j==0) return g[i][j];
        if(i<0 || j<0) return 1e9;
        if(dp[i][j]!=-1) return dp[i][j];
        int up=g[i][j]+solve(i-1,j,g,dp);
        int left=g[i][j]+solve(i,j-1,g,dp);
        return dp[i][j]=min(left,up);
    }
    int minPathSum(vector<vector<int>>& grid) {
        int m=grid.size(), n=grid[0].size();
        vector<vector<int>> dp(m,vector<int>(n,-1));
        return solve(m-1,n-1,grid,dp);
    }
    
// Tabulation....

        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(i==0 && j==0) dp[i][j]=grid[0][0];
                else{
                    int up=grid[i][j],left=grid[i][j];
                    if(i>0) up+=dp[i-1][j];
                    else up+=1e9;
                    if(j>0) left+=dp[i][j-1];
                    else left+=1e9;
                    dp[i][j]=min(up,left);
                }
            }
        }
        return dp[m-1][n-1];
    }
    
// Space Optimized

        }

    }
```

## 673. Number of Longest Increasing Subsequence

**Q:** Given an integer array `nums`, return the number of longest increasing subsequences.

**A:** 1D LIS DP extended with a secondary array to count frequencies.

**Explanation:**
- We track two states: `dp[i]` (the length of the LIS ending at `i`) and `cnt[i]` (the number of such sequences).
- When a strictly larger element is found (`nums[i] > nums[j]`), if `1 + dp[j] > dp[i]`, we found a new max length, so we update `dp[i]` and inherit `cnt[j]`.
- If `1 + dp[j] == dp[i]`, it's an alternate path of the same max length, so we add `cnt[j]` to `cnt[i]`.
- Finally, we sum all `cnt[i]` where `dp[i]` equals the global maximum LIS length `maxl`.

**Complexity:**
Tabulation: Time `O(n^2)`, Space `O(n)`

```cpp
    int findNumberOfLIS(vector<int>& nums) {
        int n=nums.size();
        int maxl=1;
        vector<int> dp(n,1),cnt(n,1);
        for(int i=1;i<n;i++){
            for(int j=i-1;j>=0;j--){
                if(nums[i]>nums[j] && 1+dp[j]>dp[i]){
                    dp[i]=1+dp[j];
                    cnt[i]=cnt[j];
                }
                else if(nums[i]>nums[j] && 1+dp[j]==dp[i]){
                    cnt[i]+=cnt[j];
                }
            }
            maxl=max(maxl,dp[i]);
        }
        int ans=0;
        for(int i=0;i<n;i++){
            if(dp[i]==maxl){
                ans+=cnt[i];
            }
        }
        return ans;
    }
```

## 70. Climbing Stairs

**Q:** Given `n` steps, return how many distinct ways you can climb to the top if you can take 1 or 2 steps at a time.

**Explanation:**
- The DP state `dp[i]` stores the distinct ways to climb `i` steps.

## 714. Best Time to Buy and Sell Stock with Transaction Fee

**Q:** Given an array of stock prices and a transaction fee, return the maximum profit you can achieve (you can complete multiple transactions but must pay the fee for each sell).

**A:** 2D State Machine DP tracking the state of holding or not holding a stock.

**Explanation:**
- The DP state `dp[i][buy]` tracks the max profit from day `i` to the end, where `buy` is a boolean indicating if we can buy (1) or sell (0).
- If we can buy, we transition to the max of buying today (`-prices[i] + dp[i+1][0]`) or skipping (`dp[i+1][1]`).
- If we can sell, we transition to the max of selling today (`prices[i] - fee + dp[i+1][1]`) or skipping (`dp[i+1][0]`).
- Base cases: when `i == n`, profit is 0.

**Complexity:**
Memoization: Time `O(n)`, Space `O(n)` + `O(n)` stack
Tabulation: Time `O(n)`, Space `O(n)`
Space Optimized: Time `O(n)`, Space `O(1)`

```cpp
// Memoization...
     int solve(int i,int buy,vector<int> &p,int &fee,vector<vector<int>> &dp){
        if(i==p.size()) return 0;
        if(dp[i][buy]!=-1) dp[i][buy];
        int ans;
        if(buy){
            ans=max(-p[i]+solve(i+1,0,p,fee,dp),solve(i+1,1,p,fee,dp));
        }else{
            ans=max(p[i]-fee+solve(i+1,1,p,fee,dp),solve(i+1,0,p,fee,dp));
        }
        return dp[i][buy]=ans;
    }

        return solve(0,1,prices,fee,dp);
```

## 72. Edit Distance

**Q:** Given two strings `word1` and `word2`, return the minimum number of operations (insert, delete, replace) required to convert `word1` to `word2`.

**A:** 2D String matching DP for string transformation.

**Explanation:**
- The DP state `dp[i][j]` tracks the minimum edit distance between the prefix of `word1` (length `i`) and prefix of `word2` (length `j`).
- If characters match (`word1[i-1] == word2[j-1]`), cost is `dp[i-1][j-1]` (no operation).
- If they don't, we take 1 plus the minimum of replacement (`dp[i-1][j-1]`), deletion (`dp[i-1][j]`), and insertion (`dp[i][j-1]`).
- Base cases: converting a string of length `i` to an empty string requires `i` deletions, and empty to length `j` requires `j` insertions.

**Complexity:**
Memorization: Time `O(n * m)`, Space `O(n * m)` + `O(n + m)` stack
Tabulation: Time `O(n * m)`, Space `O(n * m)`
Space optimize: Time `O(n * m)`, Space `O(m)`

## 823. Binary Trees With Factors

**Q:** Given an array of unique integers `arr`, return the number of binary trees we can make such that every non-leaf node's value is the product of its children's values.

**A:** 1D LIS-like DP combining subtrees formed by factors.

**Explanation:**
- The array is sorted so we can build trees bottom-up. The state `dp[i]` represents the number of valid trees with the root value `arr[i]`.
- We initialize `dp[i] = 1` for a single-node tree.
- For each `arr[i]`, we iterate over potential left children `arr[j]` (where `j < i`). If `arr[j]` divides `arr[i]` and the quotient `temp` is also in the array, we can form trees by multiplying `dp[j] * dp[mp[temp]]`.
- We use a map to quickly look up indices of factors. The answer is the sum of all `dp[i]` modulo `1e9+7`.

**Complexity:**
Tabulation: Time `O(n^2)`, Space `O(n)`

```cpp
    const int mod = 1e9 + 7;
    int numFactoredBinaryTrees(vector<int> &arr){
        vector < long long > dp(arr.size(), 1);
        unordered_map<int, int> mp;
        sort(arr.begin(), arr.end());
        
        for (int i = 0; i < arr.size(); i++){
            mp[arr[i]] = i;
            for (int j = i - 1; j >= 0; j--){
                if (arr[i] % arr[j] == 0){
                    int temp = arr[i] / arr[j];
                    if (mp.find(temp)!=mp.end()) dp[i] += (dp[j] *dp[mp[temp]]) % mod;
                }
            }
        }
        long long ans = 0;
        for(auto it : dp) ans+=it;
        return ans%mod;
    }
```

**Q:** Given an `n x n` integer matrix, return the minimum sum of any falling path through the matrix (from any element in the first row to any in the last row, moving down-left, down, or down-right).

- The answer is found at `dp[n-1]`.

```cpp
        }
```

## 26. Number of distinct subsequences

**Q:** Find the number of distinct subsequences of a given string `s` modulo `10^9+7`.

**A:** 1-D DP tracking subsequence counts with character history.

**Explanation:**
- `dp[i]` represents the total number of distinct subsequences of the prefix of length `i`.
- Adding a character doubles the subsequences, so `dp[i] = 2 * dp[i-1]`.
- If the current character appeared previously, we subtract the number of subsequences that were formed before its previous occurrence `dp[mp[s[i-1]]-1]` to avoid duplicates.
- Base case is `dp[0] = 1` for the empty subsequence.

**Complexity:**
Tabulation: Time `O(n)`, Space `O(n)`

```cpp
	        dp[i]=2*dp[i-1];
	        if(mp.find(s[i-1])!=mp.end()){
	            dp[i]-=dp[mp[s[i-1]]-1];
	        }
	        mp[s[i-1]]=i;
```

## 33. Print LCS

**Q:** Print the longest common subsequence of two given strings `x` and `y`.

**A:** 2-D DP computing the LCS table followed by backtracking.

**Explanation:**
- The function first populates a standard LCS table `dp[i][j]` with lengths of matching subsequences for prefixes of `x` and `y`.
- We backtrack from `dp[n][m]` to `dp[0][0]`. If `x[i-1] == y[j-1]`, the character belongs to the LCS and we move diagonally `(i-1, j-1)`.
- Otherwise, we move in the direction of the larger adjacent DP value (either `(i, j-1)` or `(i-1, j)`).
- The collected characters are reversed at the end since they were gathered from the end of the strings backwards.

**Complexity:**
Tabulation: Time `O(n*m)`, Space `O(n*m)`

```cpp
string printLCS(string x,string y){
    int n=x.size(), m=y.size();

    vector<vector<int>> dp(n+1,vector<int>(m+1,0));
    for(int i=1;i<n+1;i++){
        for(int j=1;j<m+1;j++){
            if(x[i-1]==y[j-1]) dp[i][j]=1+dp[i-1][j-1];
            else dp[i][j]=max(dp[i][j-1],dp[i-1][j]);
        }
    }

    string ans="";
    int i=n,j=m;
    while(i>0 && j>0){
        if(x[i-1]==y[j-1]){
            ans.push_back(x[i-1]);
            i--;j--;
        }else{
            if(dp[i][j-1]>dp[i-1][j]){
                j--;
            }else{
                i--;
            }
        }
    }
    reverse(ans.begin(),ans.end());
    return ans;
}
```

## 39. Strictly Increasing Array

**Q:** Find the minimum number of operations to make the given array `nums` strictly increasing by changing elements to any integer.

**A:** LIS-family 1-D DP tracking the maximum number of elements that can remain unchanged.

**Explanation:**
- `dp[i]` represents the length of the longest valid increasing subsequence ending at `nums[i]`.
- An element `nums[j]` can precede `nums[i]` if the difference in their values is at least the difference in their indices (`nums[i] - nums[j] >= i - j`), allowing enough integers to fill the gap strictly increasing.
- We track the overall maximum `dp[i]` as `LIS`.
- The minimum operations equal the total number of elements `n` minus the length of this valid `LIS`.

**Complexity:**
Tabulation: Time `O(n^2)`, Space `O(n)`

```cpp
int min_operations(vector<int>& nums) {
    int n = nums.size();
    vector<int> dp(n,1);
    int LIS=1;
    for(int i=1;i<n;i++){
        for(int j=0;j<i;j++){
            if(nums[i]>nums[j] && (nums[i]-nums[j])>=(i-j)){
                dp[i] = max(1+dp[j],dp[i]);
                LIS = max(LIS,dp[i]);
            }
        }
    }
    return (n-LIS);
}
```

## 40. String Subsequence

**Q:** Find the number of times `s2` appears as a subsequence in `s1`, modulo `10^9+7`.

**A:** 2-D DP tracking exact match paths.

**Explanation:**
- `dp[i][j]` represents the number of valid subsequences matching suffix `s2[j...]` in suffix `s1[i...]`.
- We always have the option to not match `s1[i]`, moving to `s1[i+1]` and keeping `s2[j]`.
- If `s1[i] == s2[j]`, we also have the option to match, moving to `s1[i+1]` and `s2[j+1]`.
- Base cases are: if `j == n2`, `s2` is fully matched returning 1; if `i == n1`, `s1` is exhausted without fully matching `s2` returning 0.

## 41. Subset Sum Problem

**Q:** Determine if there is a subset of the given array `arr` with a sum exactly equal to `sum`.

**A:** 0/1 Knapsack-family 1-D DP (space optimized) tracking achievable subset sums.

**Explanation:**
- `prev[j]` tracks whether a subset sum of `j` is possible using a prefix of the array elements.
- The state updates iteratively by checking if sum `j` was achievable without the current element, or if it is achievable by including the current element (`j - arr[i-1]`).
- Base cases are initialized such that sum 0 is always achievable (true) and any sum with 0 elements is false.
- A further space optimization processes sums backwards, requiring only one 1D array.

**Complexity:**
Memoization: Time `O(n*sum)`, Space `O(n*sum)` + `O(n)` stack
Tabulation: Time `O(n*sum)`, Space `O(n*sum)`
Space Optimized: Time `O(n*sum)`, Space `O(sum)`
More Space Optimized: Time `O(n*sum)`, Space `O(sum)`

## 43. Unbounded Knapsack

**Q:** Find the maximum total value obtained by picking any number of items with given weights `wt` and values `val`, such that their total weight doesn't exceed `W`.

**A:** Unbounded Knapsack-family 1-D DP (space optimized) allowing item reuse.

**Explanation:**
- `cur[j]` (or `prev[j]`) tracks the maximum value achievable for capacity `j`.
- The maximum value is computed by either excluding the current item, or including it (adding its value) and referencing the SAME item's state at a reduced capacity `cur[j - wt[i-1]]`.
- The most space-optimized version uses a single 1D array, iterating capacity forwards to naturally simulate item reuse.
- The overall answer is stored at capacity `W`.

**Complexity:**
Space Optimized: Time `O(N*W)`, Space `O(W)`
More Space Optimized: Time `O(N*W)`, Space `O(W)`

```cpp
// Space Optimized...
    int knapSack(int N, int W, int val[], int wt[]){
        vector<int> prev(W+1,0),cur(W+1,0);
    	for(int i=1;i<N+1;i++){
    		for(int j=1;j<W+1;j++){
    			if(j>=wt[i-1]) cur[j]=max(val[i-1]+cur[j-wt[i-1]],prev[j]);
    			else cur[j]=prev[j];
    		}
    		prev=cur;
    	}
    	return prev[W];
    }

// More Space Optimized...

    			if(j>=wt[i-1]) prev[j]=max(val[i-1]+prev[j-wt[i-1]],prev[j]);
    		}
    	}
    	return prev[W];
```

## 44. Word Break

**Q:** Determine if a string `A` can be segmented into a space-separated sequence of dictionary words given in array `B`.

**A:** 1-D DP with a hash map iterating over partition indices.

**Explanation:**
- `dp[pos]` stores whether the suffix `A[pos...]` can be segmented using words in the dictionary.
- We try every partition index `i` from `pos` to the end. If the substring `A[pos...i]` exists in the dictionary and the remaining suffix `A[i+1...]` can also be segmented, we return true.
- A hash map `dict` speeds up dictionary lookups.
- Base case is reaching the end of the string, which is naturally a valid empty segmentation returning true.

**Complexity:**
Memoization: Time `O(N^2)`, Space `O(N)` + `O(N)` stack

```cpp
            }
        }
        return dp[pos]=false;
    }
```
